SOHBET ÜZERİNDEN YAPAY ZEKA SATIŞ & FİYAT DANIŞMANI # ============================================================================== @dp.message(F.chat.type == "private", F.photo | F.text) async def ai_sales_advisor(message: types.Message): """ Kullanıcı bota bir ürün fotoğrafı veya mesaj attığında: Yapay zeka geçmiş fiyatları analiz edip güncel satış metni ve fiyat önerisi (örn: 120 TL -> 180 TL) sunar. """ if message.text and message.text.startswith("/"): return await message.answer("🔍 *Görsel ve içerik yapay zeka ile analiz ediliyor...*", parse_mode="Markdown") # AI Simülasyon Yanıtı (AI_API_KEY ile OpenAI/Gemini servisine bağlanır) suggestion = ( "🤖 **Yapay Zeka Satış & Fiyat Önerisi**\n\n" "📈 **Fiyat Analizi:** Bu ürün benzer kategoride 3 ay önce 120.00 TL'den satılmıştır. " "Güncel piyasa ve talep analizine göre önerilen yeni satış fiyatı: **160.00 TL - 180.00 TL**.\n\n" "📝 **Önerilen İkna Edici Satış Metni:**\n" "───────────────\n" "✨ *Zamana meydan okuyan tasarım! AKILLI GRUP MODERASYONU VE 15 DAKİKA SUSTURMA (MUTE) # ============================================================================== @dp.message(F.chat.type.in_({"group", "supergroup"})) async def group_moderation(message: types.Message): """ Bot sohbet gruplarında küfür, spam veya yasaklı kelime tespit ettiğinde: Mesajı siler, kullanıcıyı 15 dakika Mute eder (susturur) ve uyarı kaydı tutar. """ if not message.text: return text_lower = message.text.lower() if any(word in text_lower for word in BAD_WORDS): try: await message.delete() warn_count = await db.add_warning(message.from_user.id, message.chat.id) # 15 Dakikalık Susturma (Mute) mute_perm = ChatPermissions( can_send_messages=False, can_send_media_messages=False, can_send_other_messages=False ) until = datetime.now() + timedelta(minutes=15) await bot.restrict_chat_member( chat_id=message.chat.id, user_id=message.from_user.id, permissions=mute_perm, until_date=until ) await message.answer( f"⚠️ **AI Moderasyon Uyarısı!**\n\n" f"Kullanıcı: {message.from_user.mention_markdown()}\n" f"Sebep: Uygunsuz içerik tespiti.\n" f"Ceza: **15 Dakika Susturuldu (Mute)**\n" f"Toplam Uyarı: **{warn_count}/3**", parse_mode="Markdown" ) except Exception as e: logging.error(f"Moderasyon hatası: {e}") # ============================================================================== # 4.
